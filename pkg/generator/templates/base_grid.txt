package dayXX

import (
	"strings"

	"github.com/AnasImloul/aoc-go/pkg/registry"
	"github.com/AnasImloul/aoc-go/pkg/solver"
)

type Day struct {
	solver.Base
	grid []string
	rows int
	cols int
}

var Solver = &Day{
	Base: solver.Base{
		Year: YEAR,
		Day:  XX,
	},
}

func init() {
	Solver.FirstPart = Solver.firstPart
	Solver.SecondPart = Solver.secondPart
	Solver.ParseFunc = Solver.parse
	registry.Register(YEAR, XX, Solver)
}

func (d *Day) parse(input string) error {
	d.grid = strings.Split(strings.TrimSpace(input), "\n")
	d.rows = len(d.grid)
	if d.rows > 0 {
		d.cols = len(d.grid[0])
	}
	return nil
}

func (d *Day) inBounds(row, col int) bool {
	return row >= 0 && row < d.rows && col >= 0 && col < d.cols
}

func (d *Day) getCell(row, col int) byte {
	if !d.inBounds(row, col) {
		return 0
	}
	return d.grid[row][col]
}
